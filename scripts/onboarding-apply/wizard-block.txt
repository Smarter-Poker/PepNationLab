/**
 * Payment handles step. VERIFIED server-side, not acknowledged: the step is
 * done only when at least one non-empty handle exists on
 * agent_profiles.payment_handles. Saves through PUT /api/account/payment-method,
 * which writes profiles.payment_handles and syncs agent_profiles so the
 * dashboard's payment gate opens the moment this step completes.
 */
const WIZARD_PAYMENT_METHODS: Array<{ key: string; label: string; placeholder: string }> = [
  { key: 'zelle', label: 'Zelle', placeholder: 'Phone Number Or Email' },
  { key: 'venmo', label: 'Venmo', placeholder: '@Username' },
  { key: 'cashapp', label: 'Cash App', placeholder: '$Cashtag' },
  { key: 'apple_pay', label: 'Apple Pay', placeholder: 'Phone Number Or Apple ID Email' },
];

// Keys the payment-method API accepts. Prefilled handles outside the four
// shown above (e.g. PayPal set from Account Settings) are preserved on save.
const PAYMENT_HANDLE_KEYS = ['zelle', 'venmo', 'cashapp', 'apple_pay', 'apple_cash', 'paypal', 'google_wallet', 'wise', 'chime'];

function PaymentStep({ onDone }: { onDone: () => void }) {
  const [handles, setHandles] = useState<Record<string, string>>({});
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  useEffect(() => {
    (async () => {
      try {
        const res = await fetch('/api/account/payment-method', { cache: 'no-store' });
        const json = await res.json().catch(() => null);
        const h = (json?.data?.payment_handles ?? {}) as Record<string, string>;
        // eslint-disable-next-line react-hooks/set-state-in-effect -- one-time prefill from saved handles
        setHandles(h);
      } catch { /* prefill is best-effort */ }
    })();
  }, []);

  const set = (k: string, v: string) => setHandles((h) => ({ ...h, [k]: v }));

  const submit = async () => {
    setErr(null);
    const payload: Record<string, string> = {};
    for (const k of PAYMENT_HANDLE_KEYS) {
      const v = (handles[k] || '').trim();
      if (v) payload[k] = v;
    }
    if (WIZARD_PAYMENT_METHODS.every((m) => !payload[m.key])) {
      setErr('Add At Least One Payment Handle To Continue.');
      return;
    }
    setBusy(true);
    try {
      const res = await fetch('/api/account/payment-method', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ payment_handles: payload }),
      });
      const json = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(json.error === 'invalid_body' ? 'One Of The Handles Was Not Valid. Please Check And Try Again.' : 'Could Not Save Your Payment Methods.');
      onDone();
    } catch (e) { setErr(e instanceof Error ? e.message : 'Save Failed'); setBusy(false); }
  };

  return (
    <div>
      <StepIntro icon={CreditCard} title="Add Your Payment Methods"
        blurb="Researchers Pay You Directly Through These Handles At Checkout. Add At Least One -- You Can Add Or Change The Rest Anytime From Your Dashboard." />
      <GuidePanel steps={[
        'Fill In The Handle For At Least One Payment Method Below.',
        'Double-Check Each Handle - Researchers Send Money Straight To It.',
        'Click "Save And Continue".',
      ]} />
      {WIZARD_PAYMENT_METHODS.map((m) => (
        <Field key={m.key} label={m.label}>
          <input style={inputStyle} value={handles[m.key] ?? ''} onChange={(e) => set(m.key, e.target.value)} placeholder={m.placeholder} autoComplete="off" />
        </Field>
      ))}
      <ErrorLine msg={err} />
      <PrimaryButton onClick={submit} busy={busy}>Save And Continue</PrimaryButton>
    </div>
  );
}

/**
 * Billing step: plain-English explainer of how ordering and payment work for
 * this account's type (credit line with weekly statements vs prepaid balance).
 * Acknowledged via onboarding_progress.billing_ack.
 */
function BillingStep({ state, onDone }: { state: OnboardingState; onDone: () => void }) {
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState<string | null>(null);
  const acct = state.billing?.account_type ?? null;
  const isCredit = acct === 'credit';
  const fmt = (n: number) => `$${(Number(n) || 0).toFixed(2)}`;

  const ackNow = async () => {
    setBusy(true); setErr(null);
    try { await postOnboarding({ action: 'ack', key: 'billing' }); onDone(); }
    catch (e) { setErr(e instanceof Error ? e.message : 'Save Failed'); setBusy(false); }
  };

  return (
    <div>
      <StepIntro icon={Wallet} title="How Ordering And Credit Work"
        blurb={isCredit
          ? 'Your Account Runs On A Credit Line. Restocks And Orders Bill Against It, And You Square Up With A Weekly Statement.'
          : 'Your Account Runs On A Prepaid Balance. Restocks And Orders Deduct From It, So Keep It Topped Up.'} />

      <div style={{ padding: 'var(--space-4, 16px)', borderRadius: 10, background: 'rgba(0,196,188,0.06)', border: '1px solid rgba(0,196,188,0.22)', marginBottom: 'var(--space-4, 16px)' }}>
        <div style={{ fontSize: '0.9rem', color: 'var(--white)', marginBottom: 4 }}>{isCredit ? 'Your Credit Limit' : 'Your Prepaid Balance'}</div>
        <div style={{ fontSize: '1.8rem', fontWeight: 800, color: 'var(--teal)' }}>
          {isCredit ? fmt(state.billing?.credit_limit ?? 0) : fmt(state.billing?.prepaid_balance ?? 0)}
        </div>
      </div>

      <GuidePanel heading="What To Know" steps={isCredit
        ? [
            'When You Restock Inventory Or Approve Orders, The Wholesale Cost Bills To Your Credit Line.',
            'Every Week You Get A Statement. Open Your Wallet And Use Pay Now To Square Up By The Due Date.',
            'If An Order Is Blocked Because You Hit Your Limit, Use "Request Credit Increase" In Your Wallet - Admin Reviews It Within 1 To 2 Business Days.',
            'Click "Got It, Continue" When You Are Ready.',
          ]
        : [
            'When You Restock Inventory Or Approve Orders, The Wholesale Cost Deducts From Your Prepaid Balance.',
            'If Your Balance Cannot Cover An Order, The Order Is Held Until You Top Up.',
            'Track Your Balance And Every Transaction Anytime From Your Wallet.',
            'Click "Got It, Continue" When You Are Ready.',
          ]} />

      <ErrorLine msg={err} />
      <PrimaryButton onClick={ackNow} busy={busy}>Got It, Continue</PrimaryButton>
    </div>
  );
}

/**
 * Share step: the agent's public storefront link plus a scannable QR code
 * (reusing QRCodeGenerator). Acknowledged via onboarding_progress.share_ack.
 */
function ShareStep({ state, onDone }: { state: OnboardingState; onDone: () => void }) {
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState<string | null>(null);
  const [copied, setCopied] = useState(false);
  const slug = state.storefront?.slug ?? '';
  const url = slug ? `https://pepnationlab.com/${slug}` : 'https://pepnationlab.com';

  const copy = async () => {
    try {
      await navigator.clipboard.writeText(url);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch { /* clipboard unavailable - the link is still visible to copy manually */ }
  };

  const ackNow = async () => {
    setBusy(true); setErr(null);
    try { await postOnboarding({ action: 'ack', key: 'share' }); onDone(); }
    catch (e) { setErr(e instanceof Error ? e.message : 'Save Failed'); setBusy(false); }
  };

  return (
    <div>
      <StepIntro icon={QrCode} title="Share Your Storefront Link"
        blurb="This Is The Link Your Researchers Use To Shop With You. Share It By Text Or Email, Or Let Them Scan Your QR Code In Person." />
      <GuidePanel steps={[
        'Tap "Copy Link" To Put Your Storefront Address On The Clipboard.',
        'Send It To Your Researchers By Text, Email, Or Social.',
        'In Person? Have Them Scan The QR Code Below With Their Phone Camera.',
        'You Can Always Find This Link And QR Code Again On Your Dashboard.',
      ]} />

      <div style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '12px 14px', borderRadius: 10, background: 'rgba(255,255,255,0.04)', border: '1px solid rgba(255,255,255,0.1)', marginBottom: 'var(--space-4, 16px)', flexWrap: 'wrap' }}>
        <span style={{ flex: 1, minWidth: 180, color: 'var(--teal)', fontWeight: 700, fontSize: '0.9rem', wordBreak: 'break-all', textTransform: 'none' }}>{url}</span>
        <button type="button" className="btn btn-secondary" onClick={copy}
          style={{ display: 'inline-flex', alignItems: 'center', gap: 6, padding: '8px 14px', minHeight: 40 }}>
          {copied ? <CheckCircle2 size={15} /> : <Copy size={15} />} {copied ? 'Copied' : 'Copy Link'}
        </button>
      </div>

      <div style={{ display: 'flex', justifyContent: 'center', marginBottom: 'var(--space-4, 16px)' }}>
        <div style={{ padding: 12, borderRadius: 12, background: '#FFFFFF' }}>
          <QRCodeGenerator url={url} size={180} fgColor="#050A0F" bgColor="#FFFFFF" />
        </div>
      </div>

      <ErrorLine msg={err} />
      <PrimaryButton onClick={ackNow} busy={busy}>Got It, Continue</PrimaryButton>
    </div>
  );
}

/**
 * Compliance step: the research-use-only acknowledgment every new agent must
 * make before finishing setup. Acknowledged via
 * onboarding_progress.compliance_ack; the button stays disabled until the
 * checkbox is ticked.
 */
function ComplianceStep({ onDone }: { onDone: () => void }) {
  const [checked, setChecked] = useState(false);
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  const ackNow = async () => {
    if (!checked) return;
    setBusy(true); setErr(null);
    try { await postOnboarding({ action: 'ack', key: 'compliance' }); onDone(); }
    catch (e) { setErr(e instanceof Error ? e.message : 'Save Failed'); setBusy(false); }
  };

  return (
    <div>
      <StepIntro icon={ShieldCheck} title="Compliance Acknowledgment"
        blurb="Pep Nation Is A Research-Use-Only Platform. Before You Start Selling, Confirm You Understand The Rules Every Agent Must Follow." />
      <div style={{ display: 'flex', flexDirection: 'column', gap: 10, marginBottom: 'var(--space-4, 16px)' }}>
        <InfoRow icon={ShieldCheck} title="Research Use Only" body="Every Product Is Sold Strictly For Laboratory Research Use. Nothing On The Platform Is For Human Or Veterinary Consumption, And You Must Never Present It Otherwise." />
        <InfoRow icon={ListChecks} title="Disclaimers Stay On" body="Your Storefront Carries The Platform's Built-In Disclaimer Gates At Entry, Registration, Add-To-Cart, And Checkout. Never Ask For Them To Be Removed Or Weakened." />
        <InfoRow icon={Users} title="No Medical Claims" body="Do Not Make Dosage, Treatment, Or Health-Benefit Claims When Promoting Your Storefront. Share The Link And Let Researchers Review The Published Research Themselves." />
      </div>

      <label style={{ display: 'flex', gap: 10, alignItems: 'flex-start', padding: 'var(--space-3, 12px)', borderRadius: 10, background: 'rgba(255,255,255,0.04)', border: '1px solid rgba(255,255,255,0.1)', cursor: 'pointer', marginBottom: 'var(--space-2, 8px)' }}>
        <input type="checkbox" checked={checked} onChange={(e) => setChecked(e.target.checked)} style={{ marginTop: 3, width: 18, height: 18, accentColor: 'var(--teal)' }} />
        <span style={{ fontSize: '0.84rem', color: 'var(--grey-200, #D0DAE4)', lineHeight: 1.5 }}>
          I Understand And Agree: All Products Are Sold For Laboratory Research Use Only, Are Not For Human Consumption, And I Will Represent Them Accordingly In All Of My Selling And Marketing.
        </span>
      </label>

      <ErrorLine msg={err} />
      <PrimaryButton onClick={ackNow} busy={busy} disabled={!checked}>Agree And Continue</PrimaryButton>
    </div>
  );
}
